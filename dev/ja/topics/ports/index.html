    🔌 Port と Channel¶

    Port の種類¶

    種類 相手 recv() が返すもの
    Port 受け取るなら 1 つまで。送るだけならいくつでも 値
    MultiPort 同じ役割の相手をいくつでも connect の順の dict[Channel, 値]

    役割の違う入力（入力と教師信号、予測と誤差など）は、役割の分かる名前の別々の Port で受ける。MultiPort は、ノードが同じように束ねる同じ役割の相手のためのもの。

    class Bad(Node):
    prev = Port(recv=VECTOR)  # クラス属性: クラス文が TypeError

    def __init__(self) -> None:
        super().__init__()
        self.inputs = [Port(recv=VECTOR), Port(recv=VECTOR)]  # リスト: 実行が TypeError

    class Good(Node):
    def __init__(self) -> None:
        super().__init__()  # Port はこの後に宣言する
        self.a = Port(recv=VECTOR)  # 1 つずつ属性に代入する
        self.b = Port(recv=VECTOR)
        self.many = MultiPort(recv=VECTOR)  # 同じ役割の相手をいくつでも

    spec¶

    spec 使いどころ 初期値
    TensorSpec 活動や誤差のようなテンソル 1 つ ゼロ
    IntSpec・FloatSpec・BoolSpec・TimestampSpec 回数・フラグ・時刻 0・0.0・False・Timestamp(ns=0)
    DictSpec 入力とその目標のように一緒に届けたい値の組（TypedDict） キーごとの初期値

    回数・フラグ・時刻はテンソルに包まず、スカラーの spec で Python のスカラーとして運ぶ。受け取る側は int(...) で取り出さずにそのまま使える。

    組になる値は DictSpec で¶

    2 つの Port の値は別々に届くので、読む側は一方の新しい値ともう一方の古い値を組にして読むことがある。入力とその目標のように組になる値は 1 つの DictSpec にまとめれば、dict 全体が一度に届く。

    class Sample(TypedDict):
    x: torch.Tensor
    y: torch.Tensor


SAMPLE = DictSpec(Sample, x=TensorSpec((32, 4)), y=TensorSpec((32, 2)))

self.next = Port(send=SAMPLE)  # 送る側: self.next.send({"x": x, "y": y})
self.prev = Port(recv=SAMPLE)  # 受け取る側: sample = self.prev.recv(); sample["x"]

    型の注釈¶

    Port の型は型検査器が spec から推論する。注釈は補助関数の引数など要るときだけ Port[S, R]・MultiPort[S, R]（S を送り R を受け取る）と書き、省いた向きは Never になる（何も送らない Port の send は型検査器が拒む）。任意の spec を受け取る関数は MessageSpec で注釈する。

    def mean_of(port: MultiPort[Never, torch.Tensor]) -> torch.Tensor:
    return torch.stack(list(port.recv().values())).mean(0)

    つなぐ¶

    connect でつなぐのは Port 同士だけで、ノードとネットワークは connect を持たない。ノードは入出力が多いので、つなぐ 2 つの Port を両方書けば経路が読める。 循環（mujou.pc のノードは両方向の Port でそうしている）も、同じノードの Port 同士もつないでよい。実行する木の外の Port とつながっていると mujou.run が ValueError を送出する。

    fan-out と合流¶

    flowchart LR
    IN(["入力 x"]) --> SH["shift<br/>x + 1"]
    SH -->|"同じ値を両方へ"| D["double<br/>2x"]
    SH --> SQ["square<br/>x²"]
    D --> SUM["sum（MultiPort）<br/>Channel ごとの最新の値を足す"]
    SQ --> SUM
    SUM --> OUT(["出力 y = 2(x + 1) + (x + 1)²"])
    class Apply(Node):
    """Sends ``fn(x)``."""

    def __init__(self, fn: Callable[[torch.Tensor], torch.Tensor]) -> None:
        super().__init__()
        self.fn = fn
        self.prev = Port(recv=VECTOR)
        self.next = Port(send=VECTOR)  # only sends: any number of peers

    @override
    def on_update(self, dt: float) -> None:
        super().on_update(dt)
        self.next.send(self.fn(self.prev.recv()))


class Sum(Node):
    """Sends the sum of the latest messages of all its peers."""

    def __init__(self) -> None:
        super().__init__()
        self.prev = MultiPort(recv=VECTOR)  # merges any number of peers
        self.next = Port(send=VECTOR)

    @override
    def on_update(self, dt: float) -> None:
        super().on_update(dt)
        received = self.prev.recv()  # dict[Channel, torch.Tensor], in connect order
        self.next.send(torch.stack(list(received.values())).sum(dim=0))


class Diamond(mujou.Network):
    """``shift -> (double, square) -> sum``: y = 2 (x + 1) + (x + 1)^2."""

    def __init__(self) -> None:
        super().__init__()
        self.shift = Apply(lambda x: x + 1)
        self.double = Apply(lambda x: 2 * x)
        self.square = Apply(lambda x: x**2)
        self.sum = Sum()
        self.shift.next.connect(self.double.prev)  # one port, two peers
        self.shift.next.connect(self.square.prev)
        self.double.next.connect(self.sum.prev)  # two peers, one MultiPort
        self.square.next.connect(self.sum.prev)
        self.prev, self.next = self.shift.prev, self.sum.next

    値をどう束ねるか（足す、平均する、相手ごとに重みを掛ける）は受け取るノードが決める。

    相手ごとに扱いを変える¶

    相手ごとに扱いを変えるときは、connect が返した Channel を受け取るノードの属性に持たせ、recv() の dict のキーにする。

    # ネットワーク
self.sum.from_double = self.double.next.connect(self.sum.prev)

# 受け取るノード（sum）
x = self.prev.recv()[self.from_double]

    1 つの Channel に送る¶

    flowchart LR
    IN(["入力 x"]) --> SP["split（MultiPort）"]
    SP -->|"send_to(plus, x)"| P["pos<br/>relu"]
    SP -->|"send_to(minus, -x)"| N["neg<br/>relu"]
    P --> SUM["sum"]
    N --> SUM
    SUM --> OUT(["出力 y = relu(x) + relu(-x)<br/>（x の絶対値）"])
    class Split(Node):
    """Sends ``x`` on the channel ``plus`` and ``-x`` on the channel ``minus``."""

    # Channels of self.next, set by the network to the ones connect returned.
    plus: Channel
    minus: Channel

    def __init__(self) -> None:
        super().__init__()
        self.prev = Port(recv=VECTOR)
        self.next = MultiPort(send=VECTOR)  # sends to each peer on its own

    @override
    def on_update(self, dt: float) -> None:
        super().on_update(dt)
        x = self.prev.recv()
        self.next.send_to(self.plus, x)
        self.next.send_to(self.minus, -x)


class Magnitude(mujou.Network):
    """``split -> (relu, relu) -> sum``: y = relu(x) + relu(-x) = |x|."""

    def __init__(self) -> None:
        super().__init__()
        self.split = Split()
        self.pos, self.neg = Apply(torch.relu), Apply(torch.relu)
        self.sum = Sum()
        # Hand split the channels it sends on: the order of connect does not matter.
        self.split.plus = self.split.next.connect(self.pos.prev)
        self.split.minus = self.split.next.connect(self.neg.prev)
        self.pos.next.connect(self.sum.prev)
        self.neg.next.connect(self.sum.prev)
        self.prev, self.next = self.split.prev, self.sum.next

    send_to に渡す Channel は、ネットワークが connect の戻り値をノードの属性に代入して渡す。Channel を持つ属性は状態に入らない。

    # connect の順を入れ替えると、エラーも出ずに plus と minus が入れ替わる
self.split.plus, self.split.minus = self.split.next.channels

    self.split.plus = self.split.next.connect(self.pos.prev)
self.split.minus = self.split.next.connect(self.neg.prev)

    新しい値か読み直しか¶

    value, t = self.prev.recv_with_time()
if t is not None and t != self.last_t:  # 前回から新しい値が届いた
    self.last_t = t

    送り手が最初に送るまで、recv_with_time は初期値と時刻 None を返す。ホストが書く Port の初期値は 初期値 にある。
