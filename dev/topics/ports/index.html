    🔌 Ports and Channels¶

    The kinds of ports¶

    Kind Peers recv() returns
    Port At most one if it receives; any number if it only sends The value
    MultiPort Any number, of one role A dict[Channel, value], in connect order

    Inputs of different roles, such as an input and its teacher signal, or a prediction and an error, go to separate Ports, each named after its role. A MultiPort is for peers of one role that the node combines the same way.

    class Bad(Node):
    prev = Port(recv=VECTOR)  # class attribute: the class statement raises TypeError

    def __init__(self) -> None:
        super().__init__()
        self.inputs = [Port(recv=VECTOR), Port(recv=VECTOR)]  # list: the run raises TypeError

    class Good(Node):
    def __init__(self) -> None:
        super().__init__()  # declare ports after this
        self.a = Port(recv=VECTOR)  # one attribute per port
        self.b = Port(recv=VECTOR)
        self.many = MultiPort(recv=VECTOR)  # any number of peers of one role

    Specs¶

    Spec Used for Initial value
    TensorSpec One tensor, such as an activity or an error Zeros
    IntSpec, FloatSpec, BoolSpec, TimestampSpec A count, a flag, a time 0, 0.0, False, Timestamp(ns=0)
    DictSpec Values to deliver together, such as an input and its target (a TypedDict) The initial value per key

    A count, a flag or a time travels as a Python scalar with a scalar spec, not wrapped in a tensor, so the receiver uses it without int(...).

    Values that belong together: DictSpec¶

    Values on two ports arrive separately, so a reader may pair a new value on one with an old value on the other. Put values that belong together, such as an input and its target, in one DictSpec, and the whole dict arrives at once.

    class Sample(TypedDict):
    x: torch.Tensor
    y: torch.Tensor


SAMPLE = DictSpec(Sample, x=TensorSpec((32, 4)), y=TensorSpec((32, 2)))

self.next = Port(send=SAMPLE)  # in the sender: self.next.send({"x": x, "y": y})
self.prev = Port(recv=SAMPLE)  # in the receiver: sample = self.prev.recv(); sample["x"]

    Type annotations¶

    Type checkers infer the type of a port from its specs. Annotate only when you need to (for an argument of a helper, say), as Port[S, R] or MultiPort[S, R] (sends S, receives R); a direction left out is Never, so a type checker rejects send on a port that sends nothing. Annotate a function that takes any spec with MessageSpec.

    def mean_of(port: MultiPort[Never, torch.Tensor]) -> torch.Tensor:
    return torch.stack(list(port.recv().values())).mean(0)

    Connecting¶

    connect joins ports only: nodes and networks have no connect of their own. A node may have many inputs and outputs, and naming both ports makes the path readable. Cycles (the nodes of mujou.pc make them with two-way ports) and two ports of the same node are allowed. A port connected to one outside the tree being run makes mujou.run raise ValueError.

    Fan-out and merging¶

    flowchart LR
    IN(["Input x"]) --> SH["shift<br/>x + 1"]
    SH -->|"same value to both"| D["double<br/>2x"]
    SH --> SQ["square<br/>x²"]
    D --> SUM["sum (MultiPort)<br/>adds the latest value of each channel"]
    SQ --> SUM
    SUM --> OUT(["Output y = 2(x + 1) + (x + 1)²"])
    class Apply(Node):
    """Sends ``fn(x)``."""

    def __init__(self, fn: Callable[[torch.Tensor], torch.Tensor]) -> None:
        super().__init__()
        self.fn = fn
        self.prev = Port(recv=VECTOR)
        self.next = Port(send=VECTOR)  # only sends: any number of peers

    @override
    def on_update(self, dt: float) -> None:
        super().on_update(dt)
        self.next.send(self.fn(self.prev.recv()))


class Sum(Node):
    """Sends the sum of the latest messages of all its peers."""

    def __init__(self) -> None:
        super().__init__()
        self.prev = MultiPort(recv=VECTOR)  # merges any number of peers
        self.next = Port(send=VECTOR)

    @override
    def on_update(self, dt: float) -> None:
        super().on_update(dt)
        received = self.prev.recv()  # dict[Channel, torch.Tensor], in connect order
        self.next.send(torch.stack(list(received.values())).sum(dim=0))


class Diamond(mujou.Network):
    """``shift -> (double, square) -> sum``: y = 2 (x + 1) + (x + 1)^2."""

    def __init__(self) -> None:
        super().__init__()
        self.shift = Apply(lambda x: x + 1)
        self.double = Apply(lambda x: 2 * x)
        self.square = Apply(lambda x: x**2)
        self.sum = Sum()
        self.shift.next.connect(self.double.prev)  # one port, two peers
        self.shift.next.connect(self.square.prev)
        self.double.next.connect(self.sum.prev)  # two peers, one MultiPort
        self.square.next.connect(self.sum.prev)
        self.prev, self.next = self.shift.prev, self.sum.next

    The receiving node decides how to combine the values: sum them, average them, weight each peer.

    Treating peers differently¶

    To treat peers differently, keep the Channel that connect returned as an attribute of the receiving node, and use it as a key into the dict of recv().

    # the network
self.sum.from_double = self.double.next.connect(self.sum.prev)

# the receiving node (sum)
x = self.prev.recv()[self.from_double]

    Sending to one channel¶

    flowchart LR
    IN(["Input x"]) --> SP["split (MultiPort)"]
    SP -->|"send_to(plus, x)"| P["pos<br/>relu"]
    SP -->|"send_to(minus, -x)"| N["neg<br/>relu"]
    P --> SUM["sum"]
    N --> SUM
    SUM --> OUT(["Output y = relu(x) + relu(-x)<br/>(absolute value of x)"])
    class Split(Node):
    """Sends ``x`` on the channel ``plus`` and ``-x`` on the channel ``minus``."""

    # Channels of self.next, set by the network to the ones connect returned.
    plus: Channel
    minus: Channel

    def __init__(self) -> None:
        super().__init__()
        self.prev = Port(recv=VECTOR)
        self.next = MultiPort(send=VECTOR)  # sends to each peer on its own

    @override
    def on_update(self, dt: float) -> None:
        super().on_update(dt)
        x = self.prev.recv()
        self.next.send_to(self.plus, x)
        self.next.send_to(self.minus, -x)


class Magnitude(mujou.Network):
    """``split -> (relu, relu) -> sum``: y = relu(x) + relu(-x) = |x|."""

    def __init__(self) -> None:
        super().__init__()
        self.split = Split()
        self.pos, self.neg = Apply(torch.relu), Apply(torch.relu)
        self.sum = Sum()
        # Hand split the channels it sends on: the order of connect does not matter.
        self.split.plus = self.split.next.connect(self.pos.prev)
        self.split.minus = self.split.next.connect(self.neg.prev)
        self.pos.next.connect(self.sum.prev)
        self.neg.next.connect(self.sum.prev)
        self.prev, self.next = self.split.prev, self.sum.next

    The network hands send_to its channels by assigning what connect returned to attributes of the node. An attribute that holds a Channel is not part of the state.

    # swapping the connect calls silently swaps plus and minus
self.split.plus, self.split.minus = self.split.next.channels

    self.split.plus = self.split.next.connect(self.pos.prev)
self.split.minus = self.split.next.connect(self.neg.prev)

    A new value, or the same one again¶

    value, t = self.prev.recv_with_time()
if t is not None and t != self.last_t:  # a new value arrived since the last read
    self.last_t = t

    Until the sender first sends, recv_with_time returns the initial value with the time None. The initial values of ports the host writes are in Initial values.
